\section{Motivation: Limits of Computations}

\begin{frame}{Stone-Weierstrass: A computational theorem?!}
  \begin{itemize}
  \item \textbf{Computations} involving real numbers hinge on evaluating certain basic \textbf{continuous} operations (e.g., arithmetic) to within tolerances specified in advance.
  \item Basic operations are implemented by a \emph{floating-point library}.
    \begin{itemize}
    \item \alert{\textbf{Caveat!}} Any floating point library implements a (\emph{digital!}) algorithm which \emph{only} approximates real-valued arithmetic on \emph{bounded} intervals:
      each library only works on a \alert{specific \textbf{compact} subset of~$\mathbb{R}$}.
    \end{itemize}
  \end{itemize}
  \begin{theorem}[Stone-Weierstrass —a computational view]
    Given: (\textbf{1})~a continuous map $\alert{f: V\to \mathbb{R}}$ on a \alert{finite-dimensional} real vector space~$V$,  (\textbf{2})~a \alert{compact subset $K\subseteq V$} and (\textbf{3})~\alert{$\varepsilon>0$} arbitrary, there is an algorithm $\alert{\mathbf{A}} = \alert{\mathbf{A}_{f, K, \varepsilon}}$ which $\varepsilon$-uniformly approximates $f$ on~$K$ in the sense that $f(x)$ is $\varepsilon$-approximately evaluated relying only on composition of arithmetic operations, applied to the \alert{coordinates} of~$x$ (in some basis of~$V$), as implemented by some fixed-precision floating-point library.
  \end{theorem}
  \footnotesize{\alert{\textbf{Note:}} The \emph{Universal Approximation Theorem} for neural networks is a strengthening of Stone-Weierstrass narrowing the operations the algorithm $\alert{\mathbf{A}}$ is allowed to compose.}
\end{frame}

\begin{frame}{What does a choice of algorithm depend on?}
\alert{Take-homes} from last slide:
\begin{itemize}
\item An \alert{algorithm} to approximate a function on a vector space~$V$ depends on the choices of:
  \begin{itemize}
  \item a \alert{precision $\varepsilon>0$};
  \item the \alert{compactum $K\subseteq V$} on which the function is to be $\varepsilon$-approximated;
  \item the \alert{coordinatization of~$V$}.
  \end{itemize}
\end{itemize}
 There are many circumstances where a function is naturally defined \textbf{only} on some \alert{subset $L\subseteq V$} of a real vector space~$V$.
 \begin{itemize}
 \item E.g., $\alert{\log}$ is defined only on \alert{$(0,+\infty)\subseteq\mathbb{R}$}.
   \begin{itemize}
   \item We wish to be able to compute such functions!
   \end{itemize}
 \end{itemize}
\end{frame}

\section{Computation Structures}

\begin{frame}{Real-valued structures and computation states}
  \begin{definition}[Computation States Structure (CSS)]
    % Any real function~$P$ on a set $L$ is called a \emph{real predicate on~$L$}.
    A \emph{Computation states structure with countably many predicates} is of the form
    \begin{equation*}
      \alert{\underline{\mathbf{L}}} = \langle \alert{L}, \alert{P_i}\rangle_{i\in \mathbb{N}},
    \end{equation*}
    where
    \begin{itemize}
    \item $\alert{L}$ is a \alert{nonempty topological space};
    \item $\alert{P_i}$ is a continuous real function on $L$ (called a \emph{structural predicate}) for each $i\in \mathbb{N}$.,
    \end{itemize}
    and such that the following \textbf{\alert{faithful embedding axiom}} is satisfied:
    \begin{center}
      \emph{The \alert{\textbf{type map}} $\alert{\tp}: L\to \mathbb{R}^{\mathbb{N}}: x\mapsto \alert{\bigl(P_i(x)\bigr)_{i\in \mathbb{N}}}$ is a homeomorphic embedding into the product space~$\mathbb{R}^{\mathbb{N}}$.}
    \end{center}
    % The predicates $P_i$ of $\underline{\mathbf{L}}$ are called \emph{structural} or \emph{distinguished}.
  \end{definition}
 One may identify $L$ with its type image \alert{$\tp[L]\subseteq R^{\mathbb{N}}$}.
 \begin{itemize}
 \item The predicate $P_i$ becomes the projection $x\mapsto x_i$.
 \end{itemize}
\end{frame}

\begin{frame}{Remarks on computation states structures}
 A CSS $\underline{\mathbf{L}}$ carries the \alert{topology of coordinate-wise convergence}.
 \begin{itemize}
 \item A sequence/net in~$L$, or filter on~$L$, converges to $x\in L$ iff the pushforward of the sequence/net/filter by $P_i$ converges to $P_i(x)$, for all~$i$.
 \end{itemize}

 % A CSS is basically a Tychonoff (i.e., completely regular Hausdorff) space~$L$ with an “explicit coordinatization” by countably many real functions generating the topology of~$L$.  Therefore, many spaces of interest in real analysis are CSSs (for instance, any Polish space).
 % \begin{itemize}
 % \item The purely relational nature of CSSs does not allow endowing CSSs with operations; only the intrinsic topology on states is explicitly captured.
 % \end{itemize}

 The model-theoretic (structural) perspective shines light on:
 \begin{enumerate}
 \item \alert{Theories}: Relationships between predicate values,
   \begin{itemize}
   \item such theories may be at the heart of specific algorithms;
   \end{itemize}
 \item \alert{Explicit definability} of specific real functions on the states space,
   \begin{itemize}
   \item definability is the model-theoretic notion capturing the existence of algorithms.
   \end{itemize}
 \end{enumerate}
\end{frame}

\begin{frame}{Deep Equilibria of Neural Networks}
\begin{columns}
  \begin{column}{0.4\textwidth}
    \includegraphics[scale=0.085]{../images/deep-equilibrium-labeled}
  \end{column}
  \begin{column}{0.6\textwidth}
    \begin{itemize}
    \item A \alert{“deep state”} of $(\underline{\mathbf{L}}, T)$ is any function~$\tau: L\to L$ arising as pointwise (ultra)limit of maps $T^n$ as $n\to\infty$.
      \begin{itemize}
      \item Some such deep state $\tau\in L^L$ often arise (only) as \emph{ultra}limits, i.e., as \emph{accumulation points} of $\{T^n\}_{n\in \mathbb{N}}$ in the product topology, called \alert{ultracomputations}.
      \end{itemize}
    \end{itemize}
    \vspace{-0.3cm}
    \begin{theorem}[Alva, \textbf{D}, Iovino, Walton ’24]
      If $\underline{\mathbf{L}}$ is \emph{realcompact} and, for each fixed~$i$, the iterates set $\{P_i\circ T^n\}_{n\in \mathbb{N}}$ is bounded on compacta $K\subseteq L$, then $(\underline{\mathbf{L}}, T)$ has \alert{deep equilibria}, i.e., \textbf{idempotent} ultracomputations $\widetilde{T}: L\to L$.
    \end{theorem}
    \textbf{Proof:} Ellis-Numakura Lemma.
  \end{column}
\end{columns}
\end{frame}

\section{Classification of Deep Computations}

    \begin{block}{$\textcolor{blue}{\mathrm{B}_1}(X, Y)$: Baire Class 1 functions $X\to Y$}

    % (\emph{Real}-valued, unless otherwise specified.)
  \end{block}

\begin{frame}{Baire-1 functions \& NIP}
  Deep computations arise as limits. But how wild are these limits?

  \begin{theorem}[Fundamental Dichotomy (BFT / NIP)]
    Let $\Delta\subseteq \Cp(L, L)$ be a set of continuous transformations of a \emph{realcompact} CSS~$\underline{\mathbf{L}}$.
    The following are equivalent:
    \begin{enumerate}
    \item $\overline{\Delta}$ is \alert{relatively compact} in $\mathrm{B}_1(L, L)$ (\alert{\textbf{tameness}}).
    \item $\Delta$ has the \alert{No Independence Property (\textbf{NIP})}.
    \end{enumerate}
    If the above conditions hold, then every deep computation $\tau\in \overline{\Delta}$ is the pointwise limit of some sequence in $\overline{\Delta}$ over arbitrary compacta $K\subseteq L$.
  \end{theorem}
    
  \footnotesize{\textbf{NIP (intuition)}: $\Delta$ cannot not “shatter” arbitrarily large finite sets of inputs.}

  \footnotesize{\emph{Discrete analogue:} Finiteness of \alert{Vapnis-Chervonenkis (VC) dimension}.}
\end{frame}

\begin{frame}{Ultracomputations: From tame to wild}
  If a system satisfies NIP, it is tame.
  But there are \alert{\emph{levels} of tameness}.
  \begin{itemize}
  \item \textbf{Todorčević’s Trichotomy} for Rosenthal Compacta (compact subsets of $B_1(X)$) implies \alert{complexity classes} of ultracomputations.
    \begin{itemize}
    \item Philosophically at least, ultracomputations \emph{not} Baire-1 are \alert{wild}:
    they cannot be approximated by any algorithm relying solely on floating-point arithmetic over any compactum $K\subseteq L$ witnessing the failure of NIP.
    —i.e., wild/non-Baire-1 ultracomputations are largely incomputable over the entire states space $L$ in any reasonable sense.
    \item A Julia set is an example of a “bad” such compactum.
    \end{itemize}
  \end{itemize}
  \alert{\textbf{Remark:}}
  Glasner-Megrelishvili’s results parallel some of ours.
  Important technical differences include:
  \begin{itemize}
  \item Our framework applies to functions on the class of states spaces that are Tychonoff;
    our strongest results assume only realcompactness (rather than compact metrizability).
  \item We emphasize connections with the theory of real-valued computing.
  \end{itemize}
\end{frame}

\begin{frame}{The Seven Prototypes (ADK Heptachotomy)}
  Argyros, Dodos, and Kanellopoulos identified \emph{7 minimal building blocks} for Rosenthal compacta.
  Any wilder behavior must contain one of these prototypes.
    
    \begin{columns}
        \column{0.4\textwidth}
        \begin{enumerate}
            \item $D_1$: Alexandrov compactification (Discrete)
            \item $D_2$: One-point compactification
            \item $D_3$: Split Cantor
            \item $\dots$
            \item $D_7$: Extended split Cantor
        \end{enumerate}
        
        \column{0.6\textwidth}
        \begin{alertblock}{Implication for Computing}
        If a deep computation system satisfies NIP, its asymptotic behavior is effectively approximated by a \textbf{countable discrete set} of "archetypal" limits.
        \end{alertblock}
    \end{columns}
\end{frame}
